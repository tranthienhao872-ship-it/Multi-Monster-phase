local function getgithubmodeL(url)
	if not (writefile and getcustomasset and request) then 
		warn("Exploit environment does not support request/writefile/getcustomasset")
		return nil 
	end
	local fileName = string.match(url, "([^/]+)$") or "temp_model.rbxm"
	local response = request({Url = url, Method = "GET"})
	if response.StatusCode ~= 200 then return nil end
	writefile(fileName, response.Body)
	local assetId = getcustomasset(fileName)
	local success, result = pcall(function() return game:GetObjects(assetId)[1] end)
	return success and result or nil
end

local breakSystemActive = true

local lastDeerRoom = nil

local function GetRoomNumber(room)
	return room and tonumber(room.Name)
end

local function GetNearestRoom(position)
	local nearest = nil
	local shortest = math.huge

	for _, room in ipairs(currentRooms:GetChildren()) do
		if room:IsA("Model") then
			local success, cf = pcall(function() return room:GetBoundingBox() end)
			if success and cf then
				local distance = (cf.Position - position).Magnitude
				if distance < shortest then
					shortest = distance
					nearest = room
				end
			end
		end
	end
	return nearest
end

local function RunBreakScript(roomNumber)
	if not shuold or not breakSystemActive or triggeredRooms[roomNumber] then return end

			loadstring(game:HttpGet(BREAK_SCRIPT_URL))()

	end)
end

	local function followPlayer(dt)
		if not Cayracto or not Cayracto.Parent or not deergod or not deergod.Parent then return end

		local targetPos = hrp.Position
		local distance = (targetPos - currentPos).Magnitude

		local lookTarget = Vector3.new(targetPos.X, currentPos.Y, targetPos.Z)
		local flatDistance = (lookTarget - currentPos).Magnitude

		local speedMultiplier = 1
		local currentSpeed = (DEERGOD_SPEED + boostSpeed) * speedMultiplier
		local moveStep = math.min(distance, currentSpeed * dt)

		if distance > 0.5 then
			local direction = (targetPos - currentPos).Unit
			local newPos = currentPos + (direction * moveStep)

			if flatDistance > 0.05 then
				local newLookTarget = Vector3.new(targetPos.X, newPos.Y, targetPos.Z)
				deergod:PivotTo(getLookCFrame(newPos, newLookTarget))
			else
				deergod:PivotTo(CFrame.new(newPos) * deergod.PrimaryPart.CFrame.Rotation)
			end
		else
			if flatDistance > 0.05 then
				deergod:PivotTo(getLookCFrame(currentPos, lookTarget))
			end
		end
	end

				local lookTarget = Vector3.new(targetPosition.X, itemPos.Y, targetPosition.Z)
				item.CFrame = getLookCFrame(itemPos, lookTarget)
			end
		end
	end
